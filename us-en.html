Tampermonkey Userscript

globo_keepalive_v9.1.user.js
Interval: 3m 30s 
// ==UserScript==
// @name         Keep Alive GLOBO - Confirmed Working
// @namespace    http://tampermonkey.net/
// @version      9.1
// @description  Keeps GLOBO queue active by toggling Assignments tabs and querying call history
// @match        https://globohq.com/linguist_dashboard/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    // 3 minutes and 30 seconds interval (210,000 ms) — well before the 10-minute timeout
    const INTERVAL_MS = 210000;

    function triggerKeepAlive() {
        const currentTime = new Date().toLocaleTimeString();
        console.log(`%c[GLOBO Keep-Alive] Presence pulse initiated at: ${currentTime}`, 'color: #00bcd4; font-weight: bold;');

        // 1. Silent background network presence ping (My Video Call History)
        const historyLink = Array.from(document.querySelectorAll('a, span')).find(el => {
            const txt = (el.textContent || '').trim().toLowerCase();
            return txt.includes('my video call history') || txt.includes('my telephone call history');
        });

        if (historyLink && historyLink.closest('a')) {
            const targetUrl = historyLink.closest('a').href;
            fetch(targetUrl, { credentials: 'include' })
                .then(() => console.log('%c[GLOBO Keep-Alive] History pinged via network.', 'color: #4caf50;'))
                .catch(() => {});
        } else {
            fetch(window.location.href, { credentials: 'include' }).catch(() => {});
        }

        // 2. Safe native DOM interactions on 'Assignments' tabs (Tomorrow -> Today)
        const tabs = Array.from(document.querySelectorAll('a, button, li'));
        const tomorrowTab = tabs.find(el => (el.textContent || '').trim().toLowerCase() === 'tomorrow');
        const todayTab = tabs.find(el => (el.textContent || '').trim().toLowerCase() === 'today');

        if (tomorrowTab && todayTab) {
            tomorrowTab.click();
            console.log('[GLOBO Keep-Alive] "Tomorrow" tab triggered.');

            setTimeout(() => {
                todayTab.click();
                console.log('%c[GLOBO Keep-Alive] Return to "Today" tab completed.', 'color: #4caf50; font-weight: bold;');
            }, 1500);
        }
    }

    // First automatic test 5 seconds after page loads
    setTimeout(triggerKeepAlive, 5000);

    // Repeat on scheduled interval
    setInterval(triggerKeepAlive, INTERVAL_MS);
})();
